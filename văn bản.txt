        function playOpenSound() {
            try {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (!AudioCtx) return;
                const ctx = new AudioCtx();
                if (ctx.state === 'suspended') {
                    ctx.resume();
                }

                const notes = [523.25, 659.25, 783.99, 987.77, 1046.50]; // C5, E5, G5, B5, C6
                const now = ctx.currentTime;

                notes.forEach((freq, idx) => {
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(freq, now + idx * 0.08);

                    gain.gain.setValueAtTime(0.001, now + idx * 0.08);
                    gain.gain.exponentialRampToValueAtTime(0.18, now + idx * 0.08 + 0.02);
                    gain.gain.exponentialRampToValueAtTime(0.0001, now + idx * 0.08 + 0.55);

                    osc.connect(gain);
                    gain.connect(ctx.destination);

                    osc.start(now + idx * 0.08);
                    osc.stop(now + idx * 0.08 + 0.6);
                });
            } catch (e) {
                // Bỏ qua nếu trình duyệt chặn phát âm thanh
            }
        }

                shimmerOsc.connect(shimmerGain);
                shimmerGain.connect(gainNode);
                gainNode.connect(audioCtx.destination);

                mainOsc.start(now);
                shimmerOsc.start(now);
                mainOsc.stop(now + releaseTime + 0.05);

                }
            }

        // Hàm tự động nén ảnh xuống kích thước tối ưu (< 600KB) để lưu trữ mượt mà trên Firestore
        function compressImage(file, maxWidth = 900, quality = 0.78) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = (e) => {
                    const img = new Image();
                    img.onload = () => {
                        let width = img.width;
                        let height = img.height;

                        if (width > maxWidth || height > maxWidth) {
                            if (width > height) {
                                height = Math.round((height * maxWidth) / width);
                                width = maxWidth;
                            } else {
                                width = Math.round((width * maxWidth) / height);
                                height = maxWidth;
                            }
                        }

                        const canvas = document.createElement('canvas');
                        canvas.width = width;
                        canvas.height = height;
                        const ctx = canvas.getContext('2d');
                        ctx.drawImage(img, 0, 0, width, height);

                        let compressedDataUrl = canvas.toDataURL('image/jpeg', quality);
                        if (compressedDataUrl.length > 750000) {
                            compressedDataUrl = canvas.toDataURL('image/jpeg', 0.55);
                        }
                        resolve(compressedDataUrl);
                    };
                    img.onerror = () => reject(new Error("無法讀取此圖片格式。"));
                    img.src = e.target.result;
                };
                reader.onerror = () => reject(new Error("讀取圖片檔案發生錯誤。"));
                reader.readAsDataURL(file);
            });
        }

            if (e.target === creatorModal) {
                closeCreatorModal();
            }
        });

        modalImageInput.addEventListener('change', (e) => {
            if (e.target.files && e.target.files[0]) {
                processSelectedFile(e.target.files[0]);
                e.target.value = '';
            }
        });

        dropZoneArea.addEventListener('dragover', (e) => {
            e.preventDefault();
            dropZoneArea.classList.add('border-amber-300', 'bg-rose-900/70');
        });
        dropZoneArea.addEventListener('dragleave', (e) => {
            e.preventDefault();
            dropZoneArea.classList.remove('border-amber-300', 'bg-rose-900/70');
        });
        dropZoneArea.addEventListener('drop', (e) => {
            e.preventDefault();
            dropZoneArea.classList.remove('border-amber-300', 'bg-rose-900/70');
            if (e.dataTransfer.files && e.dataTransfer.files[0]) {
                processSelectedFile(e.dataTransfer.files[0]);
            }
        });

        saveAndGenerateBtn.addEventListener('click', saveGiftAndCreateLink);
        copyLinkBtn.addEventListener('click', copyShareLink);

        previewMyGiftBtn.addEventListener('click', () => {
            closeCreatorModal();
            if (isOpened) {
                closeGiftBox();
                setTimeout(openGiftBox, 800);
            }
        });

        createAnotherBtn.addEventListener('click', () => {
            compressedBase64Image = null;
            previewSelectedImg.src = '';
            previewSelectedImg.classList.add('hidden');
            uploadPlaceholder.classList.remove('hidden');
            changeImgOverlay.classList.add('hidden');
            changeImgOverlay.classList.remove('flex');
            shareResultStep.classList.add('hidden');
            creatorFormStep.classList.remove('hidden');
        });

        // Khởi chạy ứng dụng khi tải trang
        window.addEventListener('DOMContentLoaded', loadGiftFromUrl);
